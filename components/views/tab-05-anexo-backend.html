Editar con foto nueva
Para que editar también cambie la foto, el controlador debe leer req.file (la URL que deja Cloudinary). Además, solo actualiza los campos permitidos y responde 404 si el helado no existe.
backend/controllers/heladoController.js · función editar
javascript// editar export const editar = async (req, res) => { try { const { id } = req.params; const { nombre, descripcion, precio, stock, categoria, sabor } = req.body; // Solo se actualizan los campos que llegan en la petición const actualizaciones = {}; if (nombre !== undefined) actualizaciones.nombre = nombre; if (descripcion !== undefined) actualizaciones.descripcion = descripcion; if (precio !== undefined) actualizaciones.precio = Number(precio); if (stock !== undefined) actualizaciones.stock = Number(stock); if (categoria !== undefined) actualizaciones.categoria = categoria; if (sabor !== undefined) actualizaciones.sabor = sabor; // Si se envió una imagen nueva, Cloudinary deja la URL en req.file.path if (req.file) actualizaciones.imagen_url = req.file.path; if (Object.keys(actualizaciones).length === 0) { return res.status(400).json({ error: 'No se enviaron datos para actualizar' }); } const { data, error } = await actualizarHelado(id, actualizaciones); if (error) return res.status(500).json({ error: 'Error al actualizar' }); if (!data || data.length === 0) return res.status(404).json({ error: 'Helado no encontrado' }); return res.status(200).json({ message: 'Actualizado', helado: data[0] }); } catch (error) { return res.status(500).json({ error: error.message }); } };
